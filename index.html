<config xmlns="http://org.webharvest/schema/2.1/core">
  <!-- Fetch HTML -->
  <def var="html">
    <http url="https://example.com"/>
  </def>
  
  <!-- Parse to XML -->
  <def var="page">
    <html-to-xml>
      <get var="html"/>
    </html-to-xml>
  </def>
  
  <!-- Extract Title -->
  <def var="title">
    <xpath expression="//title/text()">
      <get var="page"/>
    </xpath>
  </def>
</config>
